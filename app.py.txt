import streamlit as st
import google.generativeai as genai
from PyPDF2 import PdfReader
import os

# --- 1. ตั้งค่าพื้นฐาน ---
# ดึง API Key จากระบบ Secrets (ที่เราจะไปตั้งค่าบนเว็บ Streamlit ภายหลัง)
try:
    genai.configure(api_key=st.secrets["GOOGLE_API_KEY"])
except:
    st.error("กรุณาตั้งค่า GOOGLE_API_KEY ในระบบ Secrets ก่อน")

model = genai.GenerativeModel('gemini-1.5-flash')

# --- 2. ฟังก์ชันอ่าน PDF ทั้งหมดในโฟลเดอร์ data ---
def get_pdf_text():
    combined_text = ""
    pdf_folder = "data/"
    
    # ตรวจสอบว่ามีโฟลเดอร์ data ไหม
    if not os.path.exists(pdf_folder):
        return "ไม่พบโฟลเดอร์ข้อมูล"
        
    for filename in os.listdir(pdf_folder):
        if filename.endswith(".pdf"):
            path = os.path.join(pdf_folder, filename)
            try:
                reader = PdfReader(path)
                for page in reader.pages:
                    combined_text += page.extract_text() + "\n"
            except Exception as e:
                print(f"อ่านไฟล์ {filename} ไม่ได้: {e}")
    return combined_text

# --- 3. ส่วนหน้าตาเว็บไซต์ (Streamlit UI) ---
st.set_page_config(page_title="Community AI Bot", layout="centered")
st.title("🤖 บอทประจำชุมชน (AI Knowledge Bot)")
st.caption("ดึงข้อมูลจากไฟล์ PDF ในคลังข้อมูลมาตอบโดยเฉพาะ")

# เก็บประวัติการแชทไว้ใน Session (ทำให้คนเข้าเว็บแยกแชทกัน)
if "messages" not in st.session_state:
    st.session_state.messages = []

# แสดงข้อความเก่าที่เคยคุย
for message in st.session_state.messages:
    with st.chat_message(message["role"]):
        st.markdown(message["content"])

# ส่วนรับคำถามใหม่
if prompt := st.chat_input("ต้องการทราบอะไรเกี่ยวกับชุมชนของเรา?"):
    # บันทึกคำถามผู้ใช้
    st.session_state.messages.append({"role": "user", "content": prompt})
    with st.chat_message("user"):
        st.markdown(prompt)

    # ส่งให้ AI ประมวลผล
    with st.chat_message("assistant"):
        with st.spinner("กำลังหาคำตอบจากคลังข้อมูล..."):
            pdf_context = get_pdf_text()
            
            # สั่ง Gemini ให้ตอบตามข้อมูลที่มี
            full_prompt = f"คุณคือผู้ช่วยตอบคำถามประจำชุมชน โดยใช้ข้อมูลต่อไปนี้เท่านั้นในการตอบ: \n{pdf_context}\n\nคำถาม: {prompt}\n(หากไม่มีข้อมูลในไฟล์ ให้ตอบว่าขออภัยไม่มีข้อมูลส่วนนี้)"
            
            response = model.generate_content(full_prompt)
            st.markdown(response.text)
            st.session_state.messages.append({"role": "assistant", "content": response.text})
